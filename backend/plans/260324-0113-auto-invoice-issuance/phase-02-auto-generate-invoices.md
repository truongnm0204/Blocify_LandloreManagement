### 2. Add autoGenerateForAllMotels() method
```typescript
async autoGenerateForAllMotels(): Promise<{
  generated: number;
  skipped: number;
  failed: number;
  errors: string[];
}> {
  const now = new Date();
  const invoiceMonth = now.getMonth() + 1;
  const invoiceYear = now.getFullYear();

  // Find ACTIVE contracts where motel has autoInvoice enabled
  const contracts = await this.prisma.contract.findMany({
    where: {
      status: ContractStatus.ACTIVE,
      motel: { autoInvoice: true },
    },
    include: {
      room: { include: { motel: true } },
      representativeTenant: true,
    },
  });

  let generated = 0, skipped = 0, failed = 0;
  const errors: string[] = [];

  for (const contract of contracts) {
    // Check if this month is a billing month for this cycle
    if (!this.isBillingMonth(contract.paymentCycle, invoiceMonth)) {
      skipped++;
      continue;
    }

    try {
      // Check duplicate
      const existing = await this.prisma.invoice.findUnique({
        where: {
          contractId_invoiceMonth_invoiceYear: {
            contractId: contract.id,
            invoiceMonth,
            invoiceYear,
          },
        },
      });
      if (existing) { skipped++; continue; }

      // Reuse generate logic (internal method)
      await this.generateInternal({
        contract, invoiceMonth, invoiceYear,
        autoGenerated: true,
        userId: 'SYSTEM',
      });
      generated++;
    } catch (err) {
      failed++;
      errors.push(`Contract ${contract.contractCode}: ${err.message}`);
    }
  }

  return { generated, skipped, failed, errors };
}
```

### 3. Helper: isBillingMonth()
```typescript
private isBillingMonth(cycle: string, month: number): boolean {
  switch (cycle) {
    case 'QUARTERLY': return [1, 4, 7, 10].includes(month);
    case 'YEARLY': return month === 1;
    default: return true; // MONTHLY
  }
}
```

### 4. Refactor generate() - Extract generateInternal()
Tach logic chung tu `generate()` thanh `generateInternal()` de reuse cho ca manual va auto.

### 5. Wire up in SchedulerService
```typescript
@Cron('0 5 0 1 * *')
async handleAutoGenerate() {
  this.logger.log('Starting auto invoice generation...');
  const result = await this.invoiceService.autoGenerateForAllMotels();
  this.logger.log(`Auto-generate: ${result.generated} created, ${result.skipped} skipped, ${result.failed} failed`);
  if (result.errors.length > 0) {
    this.logger.warn(`Errors: ${result.errors.join('; ')}`);
  }
}
```
