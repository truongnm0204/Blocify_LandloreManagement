## Implementation Steps

### 1. Add autoIssueReadyInvoices() method
```typescript
async autoIssueReadyInvoices(): Promise<{
  issued: number;
  failed: number;
  errors: string[];
}> {
  // Find DRAFT invoices that were auto-generated
  const drafts = await this.prisma.invoice.findMany({
    where: {
      status: InvoiceStatus.DRAFT,
      autoGenerated: true,
      motel: { autoInvoice: true },
    },
    include: { meterReading: true },
  });

  let issued = 0, failed = 0;
  const errors: string[] = [];

  for (const invoice of drafts) {
    try {
      await this.issueInternal(invoice.id, 'SYSTEM');
      issued++;
    } catch (err) {
      failed++;
      errors.push(`Invoice ${invoice.invoiceCode}: ${err.message}`);
    }
  }

  return { issued, failed, errors };
}
```

### 2. Refactor issue() - Extract issueInternal()
Tach logic chung tu `issue()` thanh `issueInternal(id, userId)` de reuse.

### 3. Wire up in SchedulerService
```typescript
@Cron('0 0 9 * * *')
async handleAutoIssue() {
  this.logger.log('Starting auto invoice issuance...');
  const result = await this.invoiceService.autoIssueReadyInvoices();
  this.logger.log(`Auto-issue: ${result.issued} issued, ${result.failed} failed`);
  if (result.errors.length > 0) {
    this.logger.warn(`Errors: ${result.errors.join('; ')}`);
  }
}
```
